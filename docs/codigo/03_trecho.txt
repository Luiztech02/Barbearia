export function recommendReferences(shape, texture, goal) {
  let ids = [...(order[shape] || order.oval)];
  if (texture === 'curly' || texture === 'coily') ids = goal === 'classic' ? ['afro', 'curly'] : ['curly', 'afro'];else if (goal === 'classic' && ids.includes('sidepart')) ids = ['sidepart', ...ids.filter(x => x !== 'sidepart')];else if (goal === 'bold' && shape !== 'long') ids = ['taper', ...ids.filter(x => x !== 'taper')];
  return ids.map(id => ({
    ...references.find(r => r.id === id),
    reason: why[shape] + (texture === 'curly' || texture === 'coily' ? ' Preserve a textura natural e ajuste o volume ao encolhimento dos fios.' : '')
  }));
}
